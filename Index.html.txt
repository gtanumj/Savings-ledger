(function(){

const CATS=[

const MONTHS=[

const TEMPLATE_START={year:2026,month:8};

  miscIncome:0,
  incomeAdjustments:[],
  salaryExpenses:[
    {category:"Rent/Housing",note:"Rent",amount:37000},

function empty(){
  return{
    salary:0,
    otherIncome:0,
    miscIncome:0,
    incomeAdjustments:[],
    salaryExpenses:[],
    cardExpenses:[],
    liquidSavings:[],
    investments:[],
    cardBillInitialized:false
  };
}

function clone(obj){return JSON.parse(JSON.stringify(obj))}
function uid(){return "e"+Math.random().toString(36).slice(2,9)+Date.now().toString(36).slice(-4)}
function key(y,m){return y+"-"+String(m+1).padStart(2,"0")}
function normalise(v){return String(v||"").trim().toLowerCase()}

function loadDB(){
  try{return JSON.parse(localStorage.getItem(KEY)||"{}")}catch(e){return{}}
}
function persist(){
  try{localStorage.setItem(KEY,JSON.stringify(db));return true}catch(e){return false}
}
function withIds(e){
  ["incomeAdjustments","salaryExpenses","cardExpenses","liquidSavings","investments"].forEach(g=>{
    e[g]=(e[g]||[]).map(x=>({id:x.id||uid(),...x}));
  });
  return e;
}
function previousMonth(y,m){return m===0?{year:y-1,month:11}:{year:y,month:m-1}}
function cardTotal(e){return (e.cardExpenses||[]).reduce((s,x)=>s+(Number(x.amount)||0),0)}
function previousCardSpend(y,m){
  const p=previousMonth(y,m), prev=db[key(p.year,p.month)];
  return prev?cardTotal(prev):0;
}
function previousCardNote(y,m){
  const p=previousMonth(y,m);
  return MONTHS[p.month]+" card bill";
}
function addInitialCardBill(e,y,m){
  if(e.cardBillInitialized)return e;
  const has=(e.salaryExpenses||[]).some(x=>normalise(x.category)==="credit card bills");
  if(!has){
    const amount=previousCardSpend(y,m);
    if(amount>0){
      e.salaryExpenses.unshift({id:uid(),category:"Credit Card Bills",note:previousCardNote(y,m),amount,autoCardBill:true});
    }
  }
  e.cardBillInitialized=true;
  return e;
}

function runMigrationV3(){
  if(localStorage.getItem(MIGRATION_V3)==="done")return;
  let changed=false;
  const keys=Object.keys(db).sort();

  keys.forEach(k=>{
    let e=db[k];
    if(!e||typeof e!=="object")return;
    e.incomeAdjustments=e.incomeAdjustments||[];
    e.salaryExpenses=e.salaryExpenses||[];
    e.cardExpenses=e.cardExpenses||[];
    e.investments=e.investments||[];

    e.salaryExpenses.forEach(x=>{
      if(normalise(x.note)==="support to sister"){
        x.category="Rent/Housing";
        x.note="IMI";
        changed=true;
      }
    });

    const electricity=e.salaryExpenses.filter(x=>normalise(x.note)==="electricity");
    electricity.forEach(x=>{
      const exists=e.cardExpenses.some(c=>normalise(c.note)==="electricity" && Number(c.amount||0)===Number(x.amount||0));
      if(!exists)e.cardExpenses.push({id:uid(),category:"Utilities",note:"Electricity",amount:Number(x.amount)||0});
      changed=true;
    });
    if(electricity.length)e.salaryExpenses=e.salaryExpenses.filter(x=>normalise(x.note)!=="electricity");

    db[k]=e;
  });

  keys.forEach(k=>{
    const [yy,mm]=k.split("-").map(Number);
    let e=db[k];
    if(!e)return;
    if(e.cardBillInitialized!==true){
      const has=(e.salaryExpenses||[]).some(x=>normalise(x.category)==="credit card bills");
      if(!has){
        const amount=previousCardSpend(yy,mm-1);
        if(amount>0){
          e.salaryExpenses.unshift({id:uid(),category:"Credit Card Bills",note:previousCardNote(yy,mm-1),amount,autoCardBill:true});
          changed=true;
        }
      }
      e.cardBillInitialized=true;
      changed=true;
    }
    db[k]=e;
  });

  if(changed)persist();
  localStorage.setItem(MIGRATION_V3,"done");
}

function runMigrationV4(){
  if(localStorage.getItem(MIGRATION_V4)==="done")return;
  let changed=false;

  Object.keys(db).forEach(k=>{
    const e=db[k];
    if(!e||typeof e!=="object")return;
    e.liquidSavings=e.liquidSavings||[];
    e.salaryExpenses=e.salaryExpenses||[];
    e.investments=e.investments||[];

    ["monthly sip","ulip"].forEach(target=>{
      const rows=e.salaryExpenses.filter(x=>normalise(x.note)===target);
      rows.forEach(x=>{
        const exists=e.investments.some(inv=>normalise(inv.note)===target);
        if(!exists)e.investments.push({id:uid(),note:target==="monthly sip"?"Monthly SIP":"ULIP",amount:Number(x.amount)||0,fundedFrom:"Salary Account"});
      });
      const before=e.salaryExpenses.length;
      e.salaryExpenses=e.salaryExpenses.filter(x=>normalise(x.note)!==target);
      if(e.salaryExpenses.length!==before)changed=true;
    });

    db[k]=e;
  });

  if(changed)persist();
  localStorage.setItem(MIGRATION_V4,"done");
}

function runMigrationV5(){
  if(localStorage.getItem(MIGRATION_V5)==="done")return;
  let changed=false;

  Object.keys(db).forEach(k=>{
    const e=db[k];
    if(!e||typeof e!=="object")return;

    e.miscIncome=Number(e.miscIncome)||0;
    e.liquidSavings=e.liquidSavings||[];
    e.investments=e.investments||[];

    if(Number(e.dividends||0)!==0 && !e.miscIncome){
      e.miscIncome=Number(e.dividends)||0;
      changed=true;
    }
    delete e.dividends;
    delete e.savingsInterest;

    e.liquidSavings.forEach(x=>{
      if(!x.fundedFrom){
        x.fundedFrom=normalise(x.bucket)==="bank interest"?"External":"Salary Account";
        changed=true;
      }
    });

    const hasBankInterest=e.liquidSavings.some(x=>normalise(x.bucket)==="bank interest");
    if(!hasBankInterest){
      e.liquidSavings.push({id:uid(),bucket:"Bank interest",note:"",amount:6300,fundedFrom:"External"});
      changed=true;
    }

    e.investments.forEach(x=>{
      if(!x.fundedFrom){
        x.fundedFrom="Salary Account";
        changed=true;
      }
    });

    db[k]=e;
  });

  if(changed)persist();
  localStorage.setItem(MIGRATION_V5,"done");
}

function runMigrationV6(){

function eligible(y,m){return y>TEMPLATE_START.year || (y===TEMPLATE_START.year && m>=TEMPLATE_START.month)}

function loadEntry(y,m){
  const k=key(y,m);
  if(db[k])return addInitialCardBill(withIds(clone(db[k])),y,m);
  if(eligible(y,m))return addInitialCardBill(withIds(clone(TEMPLATE)),y,m);
  return addInitialCardBill(empty(),y,m);
}

function saveEntry(y,m,e){db[key(y,m)]=clone(e);return persist()}

function fmt(n){
  n=Math.round(Number(n)||0);
  const neg=n<0;
  n=Math.abs(n);
  return (neg?"-":"")+"₹"+n.toString().replace(/\B(?=(\d{3})+(?!\d))/g,",");
}

function adjustmentTotal(e){
  return (e.incomeAdjustments||[]).reduce((s,x)=>{
    const a=Number(x.amount)||0;
    return s+((x.sign||"add")==="subtract"?-a:a);
  },0);
}

function fundedTotal(rows,source){
  return (rows||[]).reduce((s,x)=>s+(x.fundedFrom===source?(Number(x.amount)||0):0),0);
}

function totals(e){
  const adjustment=adjustmentTotal(e);
  const income=(Number(e.salary)||0)+(Number(e.otherIncome)||0)+(Number(e.miscIncome)||0)+adjustment;

  const salaryExp=(e.salaryExpenses||[]).reduce((s,x)=>s+(Number(x.amount)||0),0);
  const cardExp=cardTotal(e);

  const liquidTotal=(e.liquidSavings||[]).reduce((s,x)=>s+(Number(x.amount)||0),0);
  const liquidSalary=fundedTotal(e.liquidSavings,"Salary Account");
  const liquidCard=fundedTotal(e.liquidSavings,"Credit Card");

  const investments=(e.investments||[]).reduce((s,x)=>s+(Number(x.amount)||0),0);
  const investmentSalary=fundedTotal(e.investments,"Salary Account");
  const investmentCard=fundedTotal(e.investments,"Credit Card");

  const bankInterest=(e.liquidSavings||[]).reduce((s,x)=>s+(normalise(x.bucket)==="bank interest"?(Number(x.amount)||0):0),0);

  /* Salary-funded investments are mirrored inside Salary account expenses. */
  const salaryLeftover=income-salaryExp-liquidSalary;
  const totalSavings=salaryLeftover+liquidTotal+investments;

  return{
    income,salaryExp,cardExp,liquidTotal,liquidSalary,liquidCard,
    investments,investmentSalary,investmentCard,bankInterest,
    salaryLeftover,totalSavings,rate:income?totalSavings/income:0
  };
}

    if(item)item[el.dataset.field]=el.dataset.field==="amount" ? Number(el.value)||0 : el.value;
  });
}

function bind(){
  document.getElementById("prev").onclick=()=>change(-1);
  document.getElementById("next").onclick=()=>change(1);

  ["salaryInput","otherIncomeInput","miscIncomeInput"].forEach(id=>{
    document.getElementById(id).oninput=e=>{
      currentEntry[id==="salaryInput"?"salary":id==="otherIncomeInput"?"otherIncome":"miscIncome"]=Number(e.target.value)||0;
      refreshSummary();
    };
  });

  document.getElementById("addIncome").onclick=()=>{
    sync();
    currentEntry.incomeAdjustments.push({id:uid(),sign:"add",note:"",amount:""});
    render();
  };
  document.getElementById("addSal").onclick=()=>{
    sync();
    currentEntry.salaryExpenses.push({id:uid(),category:"Miscellaneous",note:"",amount:""});
    render();
  };
  document.getElementById("addCard").onclick=()=>{
    sync();
    currentEntry.cardExpenses.push({id:uid(),category:"Miscellaneous",note:"",amount:""});
    render();
  };
  document.getElementById("addLiquid").onclick=()=>{
    sync();
    currentEntry.liquidSavings.push({id:uid(),bucket:"",note:"",amount:"",fundedFrom:"Salary Account"});
    render();
  };
  document.getElementById("addInv").onclick=()=>{
    sync();
    const inv={id:uid(),note:"",amount:"",fundedFrom:"Salary Account"};
    currentEntry.investments.push(inv);
    syncInvestmentExpense(inv);
    render();
  };

  document.querySelectorAll(".exp-del").forEach(btn=>{
    btn.onclick=()=>{
      sync();
      if(btn.dataset.group==="investments"){
        currentEntry.salaryExpenses=(currentEntry.salaryExpenses||[]).filter(x=>x.linkedInvestmentId!==btn.dataset.id);
      }
      currentEntry[btn.dataset.group]=currentEntry[btn.dataset.group].filter(x=>x.id!==btn.dataset.id);
      render();
    };
  });

  document.querySelectorAll("[data-field]").forEach(el=>{
    const update=()=>{
      const item=(currentEntry[el.dataset.group]||[]).find(x=>x.id===el.dataset.id);
      if(!item)return;
      item[el.dataset.field]=el.dataset.field==="amount"?Number(el.value)||0:el.value;

      /* Keep salary-funded investments mirrored in Salary account expenses. */
      if(el.dataset.group==="investments")syncInvestmentExpense(item);
      refreshSummary();
    };
    el.oninput=update;
    el.onchange=update;
  });

  document.getElementById("save").onclick=save;
  document.getElementById("excel").onclick=exportExcel;
  document.getElementById("export").onclick=exportBackup;
  document.getElementById("importBtn").onclick=()=>document.getElementById("file").click();
  document.getElementById("file").onchange=importBackup;
}

function syncInvestmentExpense(inv){
  currentEntry.salaryExpenses=currentEntry.salaryExpenses||[];
  let row=currentEntry.salaryExpenses.find(x=>x.linkedInvestmentId===inv.id);

  if((inv.fundedFrom||"Salary Account")==="Salary Account"){
    if(!row){
      row={id:uid(),category:"SIP/Investments",note:inv.note||"Investment",amount:Number(inv.amount)||0,linkedInvestmentId:inv.id};
      currentEntry.salaryExpenses.push(row);
    }else{
      row.category="SIP/Investments";
      row.note=inv.note||"Investment";
      row.amount=Number(inv.amount)||0;
    }
  }else if(row){
    currentEntry.salaryExpenses=currentEntry.salaryExpenses.filter(x=>x.id!==row.id);
  }
}

function refreshSummary(){
  const t=totals(currentEntry);
  const hero=document.querySelector(".hero-figure");
  const rate=document.querySelector(".hero-rate");
  if(hero){
    hero.textContent=fmt(t.totalSavings);
    hero.className="hero-figure "+(t.totalSavings>=0?"pos":"neg");
  }
  if(rate)rate.textContent=t.income?(t.rate*100).toFixed(1)+"% savings rate":"—";

  const split=document.querySelectorAll(".savings-split-row strong");
  if(split[0])split[0].textContent=fmt(t.salaryLeftover);
  if(split[1])split[1].textContent=fmt(t.liquidTotal);
  if(split[2])split[2].textContent=fmt(t.investments);

  const totalsEls=document.querySelectorAll(".section-total");
  const vals=[t.income,t.salaryExp,t.cardExp,t.liquidTotal,t.investments];
  totalsEls.forEach((el,i)=>{if(vals[i]!==undefined)el.textContent=fmt(vals[i])});
}

function save(){
  sync();
  const ok=saveEntry(viewYear,viewMonth,currentEntry);
  const s=document.getElementById("status");
  s.textContent=ok?"Saved ✓":"Could not save";
  s.style.color=ok?"var(--green)":"var(--red)";
  renderYear();
  setTimeout(()=>{if(s)s.textContent=""},2500);
}

function change(direction){
  sync();
  saveEntry(viewYear,viewMonth,currentEntry);
  viewMonth+=direction;
  if(viewMonth<0){viewMonth=11;viewYear--}
  if(viewMonth>11){viewMonth=0;viewYear++}
  currentEntry=loadEntry(viewYear,viewMonth);
  render();
}

function renderYear(){
  const wrap=document.getElementById("year");
  const rows=[];
  for(let m=0;m<12;m++){
    let e=m===viewMonth?currentEntry:(db[key(viewYear,m)]?withIds(clone(db[key(viewYear,m)])):empty());
    const t=totals(e);
    rows.push({month:MONTHS[m].slice(0,3),income:t.income,expenses:t.salaryExp,card:t.cardExp,leftover:t.salaryLeftover,liquid:t.liquidTotal,investments:t.investments,total:t.totalSavings});
  }

  const sum=f=>rows.reduce((s,r)=>s+r[f],0);
  const yi=sum("income"), ye=sum("expenses"), yc=sum("card"), yl=sum("leftover"), yliq=sum("liquid"), yinv=sum("investments"), ys=sum("total");
  const yr=yi?ys/yi*100:0;
  const max=Math.max(1,...rows.map(r=>Math.abs(r.total))),bw=26,gap=10,ch=90,base=45,w=rows.length*(bw+gap);
  const bars=rows.map((r,i)=>{
    const x=i*(bw+gap),h=Math.abs(r.total)/max*(ch/2-6),y=r.total>=0?base-h:base,c=r.total>=0?"var(--green)":"var(--red)";
    return `<rect x="${x}" y="${y}" width="${bw}" height="${Math.max(h,1)}" fill="${c}" rx="2"/><text x="${x+bw/2}" y="${ch+12}" font-size="9" fill="var(--ink-soft)" text-anchor="middle">${r.month}</text>`;
  }).join("");

  wrap.innerHTML=`
    <div class="year-kpis">
      <div class="kpi"><div class="kpi-label">Income</div><div class="kpi-value">${fmt(yi)}</div></div>
      <div class="kpi"><div class="kpi-label">Salary expenses</div><div class="kpi-value">${fmt(ye)}</div></div>
      <div class="kpi"><div class="kpi-label">Credit card spending</div><div class="kpi-value">${fmt(yc)}</div></div>
      <div class="kpi"><div class="kpi-label">Savings rate</div><div class="kpi-value">${yr.toFixed(1)}%</div></div>
    </div>
    <div class="year-kpis">
      <div class="kpi"><div class="kpi-label">Salary leftover</div><div class="kpi-value">${fmt(yl)}</div></div>
      <div class="kpi"><div class="kpi-label">Liquid savings</div><div class="kpi-value green">${fmt(yliq)}</div></div>
      <div class="kpi"><div class="kpi-label">Investments</div><div class="kpi-value gold">${fmt(yinv)}</div></div>
      <div class="kpi"><div class="kpi-label">Total savings</div><div class="kpi-value green">${fmt(ys)}</div></div>
    </div>
    <div class="chart-wrap"><svg viewBox="0 0 ${w} ${ch+20}" width="100%" height="140"><line x1="0" y1="${base}" x2="${w}" y2="${base}" stroke="var(--paper-line)"/>${bars}</svg></div>
    <table class="year-table">
      <thead><tr><th>Month</th><th>Leftover</th><th>Liquid</th><th>Invest.</th><th>Total</th></tr></thead>
      <tbody>${rows.map(r=>`<tr><td>${r.month}</td><td class="ledger-num">${fmt(r.leftover)}</td><td class="ledger-num">${fmt(r.liquid)}</td><td class="ledger-num">${fmt(r.investments)}</td><td class="ledger-num">${fmt(r.total)}</td></tr>`).join("")}
      <tr><td><strong>Total</strong></td><td class="ledger-num"><strong>${fmt(yl)}</strong></td><td class="ledger-num"><strong>${fmt(yliq)}</strong></td><td class="ledger-num"><strong>${fmt(yinv)}</strong></td><td class="ledger-num"><strong>${fmt(ys)}</strong></td></tr></tbody>
    </table>
  `;
}

function exportExcel(){
  sync();
  saveEntry(viewYear,viewMonth,currentEntry);
  if(typeof XLSX==="undefined"){alert("Excel export could not load. Please connect to the internet and try again.");return}

  const monthly=[],salaryRows=[],cardRows=[],liquidRowsX=[],investmentRowsX=[],adjustmentRows=[];
  Object.keys(db).sort().forEach(k=>{
    const [year,mm]=k.split("-").map(Number),month=MONTHS[mm-1],e=db[k]||empty(),t=totals(e);
    monthly.push({
      Year:year,Month:month,Salary:Number(e.salary)||0,"Rental / Other Income":Number(e.otherIncome)||0,
      Miscellaneous:Number(e.miscIncome)||0,"Income Adjustments":adjustmentTotal(e),"Total Income":t.income,
      "Salary Account Expenses":t.salaryExp,"Credit Card Spending":t.cardExp,"Salary Account Leftover":t.salaryLeftover,
      "Liquid Savings":t.liquidTotal,Investments:t.investments,"Total Savings":t.totalSavings,"Savings Rate %":Number((t.rate*100).toFixed(2))
    });
    (e.salaryExpenses||[]).forEach(x=>salaryRows.push({Year:year,Month:month,Category:x.category||"",Note:x.note||"",Amount:Number(x.amount)||0}));
    (e.cardExpenses||[]).forEach(x=>cardRows.push({Year:year,Month:month,Category:x.category||"",Note:x.note||"",Amount:Number(x.amount)||0}));
    (e.liquidSavings||[]).forEach(x=>liquidRowsX.push({Year:year,Month:month,Bucket:x.bucket||"",Note:x.note||"","Funded From":x.fundedFrom||"",Amount:Number(x.amount)||0}));
    (e.investments||[]).forEach(x=>investmentRowsX.push({Year:year,Month:month,Investment:x.note||"","Funded From":x.fundedFrom||"",Amount:Number(x.amount)||0}));
    (e.incomeAdjustments||[]).forEach(x=>{const a=Number(x.amount)||0;adjustmentRows.push({Year:year,Month:month,Type:(x.sign||"add")==="subtract"?"Subtract":"Add",Note:x.note||"",Amount:(x.sign||"add")==="subtract"?-a:a})});
  });

  const wb=XLSX.utils.book_new();
  [["Monthly Summary",monthly],["Salary Expenses",salaryRows],["Credit Card",cardRows],["Liquid Savings",liquidRowsX],["Investments",investmentRowsX],["Income Adjustments",adjustmentRows]].forEach(([name,data])=>{
    XLSX.utils.book_append_sheet(wb,XLSX.utils.json_to_sheet(data),name);
  });

  const d=new Date(),stamp=d.getFullYear()+"-"+String(d.getMonth()+1).padStart(2,"0")+"-"+String(d.getDate()).padStart(2,"0");
  const output=XLSX.write(wb,{bookType:"xlsx",type:"array"});
  downloadBlob(new Blob([output],{type:"application/vnd.openxmlformats-officedocument.spreadsheetml.sheet"}),"Savings_Ledger_All_Years_"+stamp+".xlsx");
}

function downloadBlob(blob,filename){
  const url=URL.createObjectURL(blob),a=document.createElement("a");
  a.href=url;a.download=filename;document.body.appendChild(a);a.click();a.remove();
  setTimeout(()=>URL.revokeObjectURL(url),1000);
}

function exportBackup(){
  sync();saveEntry(viewYear,viewMonth,currentEntry);
  downloadBlob(new Blob([JSON.stringify(db,null,2)],{type:"application/json"}),"savings-ledger-backup.json");
}

async function importBackup(e){
  const f=e.target.files[0];
  if(!f)return;
  try{
    const obj=JSON.parse(await f.text());
    if(!obj||typeof obj!=="object"||Array.isArray(obj))throw new Error();
    db=obj;persist();runMigrationV5();currentEntry=loadEntry(viewYear,viewMonth);render();
  }catch(err){alert("Invalid backup file.")}
  e.target.value="";
}
